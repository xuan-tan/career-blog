{{/* editorial/card.html — a card in the editorial grid, or the lead feature. Params: page (Page, required) variant ("card" | "feature", default "card") Structure (a card): image title date · reading time <- own line under the title; posts only summary metadata bar <- tags · entry count, between two hairlines So the piece's own facts (when, how long) sit together on one line right under the title, and the summary reads as the description beneath them. The bar is left with only the filterable things. Sections have no date or reading time, so that line simply does not render for them; their entry count stays in the bar. The feature keeps its title | standfirst two-column split, so its date · reading time sits inside the title column rather than spanning. Notes: - Grid/feature images are wrapped in a navigation link, so they carry `nozoom`. Blowfish binds medium-zoom to `img:not(.nozoom)` (see layouts/partials/footer.html) — without it the click both navigated AND opened the lightbox, which pushed a history entry and made "back" reopen the spotlight instead of the list. - Spec: .design-system.md §7 */}} {{- $page := .page -}} {{- $variant := .variant | default "card" -}} {{- $isFeature := eq $variant "feature" -}} {{- $width := cond $isFeature 1600 900 -}} {{- $img := partial "editorial/image.html" (dict "page" $page "width" $width) -}} {{- $href := $page.RelPermalink -}} {{- with $page.Params.externalUrl -}}{{- $href = . -}}{{- end -}} {{/* Tag terms (real taxonomy pages, so the links resolve through Hugo's own permalink config rather than a hand-built URL). */}} {{- $terms := $page.GetTerms "tags" -}} {{/* Post line: date then reading time, for posts only. */}} {{- $postMeta := slice -}} {{- $info := slice -}} {{- if $page.IsSection -}} {{/* Sections carry an entry count, no date/reading time. */}} {{- with len $page.RegularPagesRecursive -}} {{- $info = $info | append (printf "%d %s" . (cond (eq . 1) "entry" "entries")) -}} {{- end -}} {{- else -}} {{- if not $page.Date.IsZero -}} {{- $postMeta = $postMeta | append ($page.Date.Format "2 Jan 2006") -}} {{- end -}} {{- $postMeta = $postMeta | append (printf "%d min read" $page.ReadingTime) -}} {{- end -}} {{- $postMetaStr := delimit $postMeta " · " -}} {{- $infoStr := delimit $info " · " -}} {{- $summary := "" -}} {{- with $page.Params.description -}} {{- $summary = . -}} {{- end -}} {{- if or (eq $summary "") (eq $summary $page.Title) -}} {{- $summary = $page.Summary | plainify | strings.TrimSpace -}} {{- end -}} {{- $hasBar := or $terms $infoStr -}} {{- if $isFeature -}}
