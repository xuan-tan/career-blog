{{/* editorial/card.html — a card in the editorial grid, or the lead feature. Params: page (Page, required) variant ("card" | "feature", default "card") Structure: image title (+ reading time, inline after it) standfirst / summary metadata bar <- tags · descriptor, between two hairlines Reading time sits inline after the title because it belongs to the piece, not to the catalogue metadata. The bar keeps only things you might filter by: tags (links) and the descriptor (entry count for a section, date for a post). Why nothing sits above the title: an earlier version kept the descriptor as a kicker above the title, but only some cards have one (a section with no entries has no count), which staggered the titles by ~21px across a row. One band, always at the foot, keeps every card structurally identical and the titles aligned. Notes: - Grid/feature images are wrapped in a navigation link, so they carry `nozoom`. Blowfish binds medium-zoom to `img:not(.nozoom)` (see layouts/partials/footer.html) — without it the click both navigated AND opened the lightbox, which pushed a history entry and made "back" reopen the spotlight instead of the list. - Spec: .design-system.md §7 */}} {{- $page := .page -}} {{- $variant := .variant | default "card" -}} {{- $isFeature := eq $variant "feature" -}} {{- $width := cond $isFeature 1600 900 -}} {{- $img := partial "editorial/image.html" (dict "page" $page "width" $width) -}} {{- $href := $page.RelPermalink -}} {{- with $page.Params.externalUrl -}}{{- $href = . -}}{{- end -}} {{/* Tag terms (real taxonomy pages, so the links resolve through Hugo's own permalink config rather than a hand-built URL). */}} {{- $terms := $page.GetTerms "tags" -}} {{/* Bar descriptor: entry count for a section, date for a post. A post's reading time is NOT in here - it goes inline after the title. */}} {{- $info := slice -}} {{- $readTime := "" -}} {{- if $page.IsSection -}} {{- with len $page.RegularPagesRecursive -}} {{- $info = $info | append (printf "%d %s" . (cond (eq . 1) "entry" "entries")) -}} {{- end -}} {{- else -}} {{- if not $page.Date.IsZero -}} {{- $info = $info | append ($page.Date.Format "2 Jan 2006") -}} {{- end -}} {{- $readTime = printf "%d min read" $page.ReadingTime -}} {{- end -}} {{- $infoStr := delimit $info " · " -}} {{- $summary := "" -}} {{- with $page.Params.description -}} {{- $summary = . -}} {{- end -}} {{- if or (eq $summary "") (eq $summary $page.Title) -}} {{- $summary = $page.Summary | plainify | strings.TrimSpace -}} {{- end -}} {{- $hasBar := or $terms $infoStr -}} {{- if $isFeature -}}
